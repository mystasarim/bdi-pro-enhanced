class ExerciseVisualView extends View {
    private final Paint paint = new Paint(Paint.ANTI_ALIAS_FLAG);
    private final ValueAnimator animator;
    private int exerciseIndex = 0;
    private boolean male = true;
    private float phase = 0f;

    ExerciseVisualView(Context context) {
        super(context);
        setLayerType(LAYER_TYPE_SOFTWARE, null);
        animator = ValueAnimator.ofFloat(0f, 1f);
        animator.setDuration(1450L);
        animator.setRepeatCount(ValueAnimator.INFINITE);
        animator.setRepeatMode(ValueAnimator.REVERSE);
        animator.setInterpolator(new android.view.animation.AccelerateDecelerateInterpolator());
        animator.addUpdateListener(a -> {
            phase = (float) a.getAnimatedValue();
            invalidate();
        });
        post(animator::start);
    }

    void setExercise(int exerciseIndex, boolean male) {
        this.exerciseIndex = exerciseIndex;
        this.male = male;
        phase = 0f;
        if (!animator.isStarted()) animator.start();
        invalidate();
    }

    @Override
    protected void onAttachedToWindow() {
        super.onAttachedToWindow();
        if (!animator.isStarted()) animator.start();
    }

    @Override
    protected void onDetachedFromWindow() {
        animator.cancel();
        super.onDetachedFromWindow();
    }

    private float mix(float a, float b) { return a + (b - a) * phase; }

    private Pose blend(Pose a, Pose b) {
        Pose p = new Pose();
        for (int i = 0; i < p.v.length; i++) p.v[i] = mix(a.v[i], b.v[i]);
        return p;
    }

    @Override
    protected void onDraw(Canvas canvas) {
        super.onDraw(canvas);
        float w = getWidth();
        float h = getHeight();
        RectF area = new RectF(3, 3, w - 3, h - 3);
        Path clip = new Path();
        clip.addRoundRect(area, 34, 34, Path.Direction.CW);
        canvas.save();
        canvas.clipPath(clip);

        paint.setShader(new LinearGradient(0, 0, w, h,
                new int[]{Color.WHITE, Color.rgb(243, 250, 248), Color.rgb(231, 244, 249)},
                null, Shader.TileMode.CLAMP));
        canvas.drawRect(area, paint);
        paint.setShader(null);

        paint.setColor(Color.rgb(218, 230, 233));
        canvas.drawRoundRect(new RectF(w * .08f, h * .84f, w * .92f, h * .865f), 16, 16, paint);

        if (exerciseIndex >= 12 && exerciseIndex <= 15) {
            drawBreathingPelvic(canvas, w, h);
        } else {
            Pose[] pair = posesFor(exerciseIndex);
            drawHuman(canvas, blend(pair[0], pair[1]), w, h);
            drawMotionCue(canvas, pair[0], pair[1], w, h);
        }

        canvas.restore();
        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeWidth(2f);
        paint.setColor(Color.rgb(226, 232, 240));
        canvas.drawRoundRect(area, 34, 34, paint);
        paint.setStyle(Paint.Style.FILL);
    }

    private Pose[] posesFor(int index) {
        switch (index) {
            case 1: return new Pose[]{standing(), squat()};
            case 2: return new Pose[]{pushupTop(), pushupBottom()};
            case 3: return new Pose[]{rowExtended(), rowPulled()};
            case 4: return new Pose[]{standingWeight(), deadliftBottom()};
            case 5: return new Pose[]{bridgeBottom(), bridgeTop()};
            case 6: return new Pose[]{plankPose(), plankPose()};
            case 7: return new Pose[]{lungeTop(), lungeBottom()};
            case 8: return new Pose[]{pressBottom(), pressTop()};
            case 9: return new Pose[]{supermanBottom(), supermanTop()};
            case 10: return new Pose[]{sidePlankLow(), sidePlankHigh()};
            default: return new Pose[]{walkLeft(), walkRight()};
        }
    }

    private void drawHuman(Canvas c, Pose n, float w, float h) {
        float u = Math.min(w, h);
        int skin = male ? Color.rgb(217, 164, 126) : Color.rgb(229, 180, 144);
        int top = male ? Color.rgb(11, 29, 58) : Color.rgb(19, 133, 118);
        int bottom = Color.rgb(24, 34, 52);
        float limb = u * (male ? .035f : .031f);

        float hx = n.x(0,w), hy = n.y(0,h);
        float sx = n.x(1,w), sy = n.y(1,h);
        float hipx = n.x(2,w), hipy = n.y(2,h);
        float elx = n.x(3,w), ely = n.y(3,h), hlx = n.x(4,w), hly = n.y(4,h);
        float erx = n.x(5,w), ery = n.y(5,h), hrx = n.x(6,w), hry = n.y(6,h);
        float klx = n.x(7,w), kly = n.y(7,h), alx = n.x(8,w), aly = n.y(8,h);
        float krx = n.x(9,w), kry = n.y(9,h), arx = n.x(10,w), ary = n.y(10,h);

        segment(c, hipx, hipy, krx, kry, bottom, limb * 1.20f);
        segment(c, krx, kry, arx, ary, skin, limb);
        segment(c, sx, sy, erx, ery, skin, limb);
        segment(c, erx, ery, hrx, hry, skin, limb * .85f);

        paint.setStrokeCap(Paint.Cap.ROUND);
        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeWidth(u * (male ? .115f : .092f));
        paint.setColor(top);
        c.drawLine(sx, sy, hipx, hipy, paint);
        paint.setStyle(Paint.Style.FILL);

        segment(c, hipx, hipy, klx, kly, bottom, limb * 1.24f);
        segment(c, klx, kly, alx, aly, skin, limb);
        segment(c, sx, sy, elx, ely, skin, limb);
        segment(c, elx, ely, hlx, hly, skin, limb * .85f);

        paint.setColor(skin);
        c.drawCircle(hlx, hly, limb * .55f, paint);
        c.drawCircle(hrx, hry, limb * .55f, paint);
        c.drawCircle(elx, ely, limb * .45f, paint);
        c.drawCircle(erx, ery, limb * .45f, paint);
        c.drawCircle(klx, kly, limb * .52f, paint);
        c.drawCircle(krx, kry, limb * .52f, paint);

        paint.setColor(skin);
        c.drawCircle(hx, hy, u * .058f, paint);
        paint.setColor(Color.rgb(28, 35, 49));
        RectF hair = new RectF(hx-u*.060f, hy-u*.065f, hx+u*.060f, hy+u*.025f);
        c.drawArc(hair, 190, 160, true, paint);
        if (!male) c.drawCircle(hx + u*.052f, hy, u*.025f, paint);

        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeCap(Paint.Cap.ROUND);
        paint.setStrokeWidth(u*.022f);
        paint.setColor(Color.rgb(11,29,58));
        c.drawLine(alx-u*.018f, aly, alx+u*.045f, aly, paint);
        c.drawLine(arx-u*.018f, ary, arx+u*.045f, ary, paint);
        paint.setStyle(Paint.Style.FILL);
    }

    private void segment(Canvas c, float x1, float y1, float x2, float y2, int color, float width) {
        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeCap(Paint.Cap.ROUND);
        paint.setStrokeWidth(width);
        paint.setColor(color);
        c.drawLine(x1, y1, x2, y2, paint);
        paint.setStyle(Paint.Style.FILL);
    }

    private void drawMotionCue(Canvas c, Pose a, Pose b, float w, float h) {
        float x1 = a.x(2,w), y1 = a.y(2,h), x2 = b.x(2,w), y2 = b.y(2,h);
        if (Math.abs(x2-x1)+Math.abs(y2-y1) < 8f) return;
        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeWidth(Math.max(4f, w*.009f));
        paint.setStrokeCap(Paint.Cap.ROUND);
        paint.setColor(Color.argb(150,17,77,216));
        c.drawLine(x1, y1, mix(x1,x2), mix(y1,y2), paint);
        paint.setStyle(Paint.Style.FILL);
    }

    private void drawBreathingPelvic(Canvas c, float w, float h) {
        Pose seated = seatedPose();
        drawHuman(c, seated, w, h);
        float cx = seated.x(2,w);
        float cy = seated.y(2,h) - h*.02f;
        float pulse = .72f + phase*.28f;
        paint.setStyle(Paint.Style.STROKE);
        paint.setStrokeWidth(Math.max(5f,w*.012f));
        paint.setColor(Color.argb(185,78,215,168));
        c.drawCircle(cx, cy, w*.07f*pulse, paint);
        paint.setColor(Color.argb(105,0,194,255));
        c.drawCircle(cx, cy, w*.11f*pulse, paint);
        paint.setStyle(Paint.Style.FILL);
        paint.setTextAlign(Paint.Align.CENTER);
        paint.setTypeface(android.graphics.Typeface.DEFAULT_BOLD);
        paint.setTextSize(w*.042f);
        paint.setColor(Color.rgb(11,29,58));
        String text = exerciseIndex == 12 || exerciseIndex == 13
                ? (phase < .5f ? "Nazikçe sık" : "Tamamen gevşe")
                : (phase < .5f ? "Burundan nefes al" : "Yavaşça nefes ver");
        c.drawText(text, w*.50f, h*.16f, paint);
    }

    private Pose P(double... values) {
        Pose p = new Pose();
        for (int i=0;i<values.length && i<p.v.length;i++) p.v[i]=(float)values[i];
        return p;
    }

    private static class Pose {
        final float[] v = new float[22];
        float x(int point, float w) { return v[point*2] * w; }
        float y(int point, float h) { return v[point*2+1] * h; }
    }
}
